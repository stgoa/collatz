@LeanLane.GenPer.invariant_classification : ∀ {α : Type u_1} (h : ℕ → α) {d M : ℕ},
  Odd d →
    0 < M →
      (∀ (n : ℕ), h (n + M) = h n) →
        (∀ (n : ℕ), h (2 * n) = h n) →
          (∀ (n : ℕ), Odd n → 1 < n → h (3 * n + d) = h n) →
            (∀ (n : ℕ), h (n + M.gcd d) = h n) ∧ (∀ (n : ℕ), h (2 * n) = h n) ∧ ∀ (n : ℕ), h (3 * n) = h n

@LeanLane.GenPer.control_indicator : ∀ {d : ℕ},
  Odd d →
    ¬3 ∣ d →
      (∀ (n : ℕ), LeanLane.GenPer.indicator d (n + d) = LeanLane.GenPer.indicator d n) ∧
        (∀ (n : ℕ), LeanLane.GenPer.indicator d (2 * n) = LeanLane.GenPer.indicator d n) ∧
          (∀ (n : ℕ), Odd n → 1 < n → LeanLane.GenPer.indicator d (3 * n + d) = LeanLane.GenPer.indicator d n) ∧
            (1 < d → LeanLane.GenPer.indicator d 1 ≠ LeanLane.GenPer.indicator d d)
def LeanLane.GenPer.indicator : ℕ → ℕ → Bool :=
fun d n => decide (d ∣ n)
