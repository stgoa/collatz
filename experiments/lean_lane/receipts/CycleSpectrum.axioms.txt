LeanLane.Cyc.commutator : ∀ (u v : List ℕ) (k : ℕ),
  LeanLane.Cyc.D u * LeanLane.Cyc.C (LeanLane.Cyc.rep u k ++ v) -
      LeanLane.Cyc.C u * LeanLane.Cyc.D (LeanLane.Cyc.rep u k ++ v) =
    (2 ^ u.sum) ^ k * (LeanLane.Cyc.D u * LeanLane.Cyc.C v - LeanLane.Cyc.D v * LeanLane.Cyc.C u)

LeanLane.Cyc.gcd_periodic : ∀ (u v : List ℕ),
  (∀ a ∈ u, 1 ≤ a) →
    u ≠ [] →
      (∀ a ∈ v, 1 ≤ a) →
        v ≠ [] →
          ∀ (a b K₀ : ℕ),
            (LeanLane.Cyc.D u * LeanLane.Cyc.C v - LeanLane.Cyc.D v * LeanLane.Cyc.C u).natAbs = 2 ^ a * 3 ^ b * K₀ →
              ∀ (T : ℕ),
                (2 ^ u.sum) ^ T ≡ 1 [MOD K₀ * (LeanLane.Cyc.D u).natAbs] →
                  (3 ^ u.length) ^ T ≡ 1 [MOD K₀ * (LeanLane.Cyc.D u).natAbs] →
                    ∀ (k : ℕ),
                      (LeanLane.Cyc.C (LeanLane.Cyc.rep u (k + T) ++ v)).gcd
                          (LeanLane.Cyc.D (LeanLane.Cyc.rep u (k + T) ++ v)) =
                        (LeanLane.Cyc.C (LeanLane.Cyc.rep u k ++ v)).gcd (LeanLane.Cyc.D (LeanLane.Cyc.rep u k ++ v))
LeanLane.Cyc.gcd_periodic_totient : ∀ (u v : List ℕ),
  (∀ a ∈ u, 1 ≤ a) →
    u ≠ [] →

        v ≠ [] →

                    (LeanLane.Cyc.C (LeanLane.Cyc.rep u k ++ v)).gcd (LeanLane.Cyc.D (LeanLane.Cyc.rep u k ++ v))
LeanLane.Cyc.seven_dvd_D_example : ∀ (k : ℕ), 7 ∣ LeanLane.Cyc.D (LeanLane.Cyc.rep [1, 1, 3] k ++ [2]) ↔ k % 6 = 3
def LeanLane.Cyc.C : List ℕ → ℤ :=
fun x => List.brecOn x LeanLane.Cyc.C._f
def LeanLane.Cyc.D : List ℕ → ℤ :=
fun z => 2 ^ z.sum - 3 ^ z.length
def LeanLane.Cyc.rep : List ℕ → ℕ → List ℕ :=
fun u x => Nat.brecOn x (LeanLane.Cyc.rep._f u)
def LeanLane.Cyc.G : ℤ → ℤ → ℕ → ℤ :=
fun Q P x => Nat.brecOn x (LeanLane.Cyc.G._f Q P)
